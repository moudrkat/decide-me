cmake_minimum_required(VERSION 3.22)
project(decide_vision LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
if(ANDROID)
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
endif()
get_filename_component(LLAMA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../vendor/kneejerk/llama.cpp" ABSOLUTE)
add_subdirectory(${LLAMA_ROOT} ${CMAKE_CURRENT_BINARY_DIR}/llama EXCLUDE_FROM_ALL)
add_library(decide_vision SHARED vision.cpp)
target_include_directories(decide_vision PRIVATE ${LLAMA_ROOT}/tools/mtmd)
target_link_libraries(decide_vision PRIVATE llama mtmd)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(decide_vision PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_compile_options(decide_vision PRIVATE -fvisibility=hidden)
# Both plugins statically link llama.cpp. Keep each runtime's symbols private.
target_link_options(decide_vision PRIVATE "-Wl,--exclude-libs,ALL")
